% Opening stack y0 targets (print p.1): 89.4 / 147.6 / 207.7 / 237.1 / 281.0 / 315.6

% Strip accidental \textbf from generated heading bodies when the macro
% sets its own weight (running heads / piṭaka: medium, not bold).
\newcommand{\csromanplainarg}[1]{%
  \begingroup
  \renewcommand{\textbf}[1]{##1}%
  #1%
  \endgroup
}

\newcommand{\csromanheadingfont}[3]{%
  % #1 size bp, #2 baselineskip bp, #3 text
  \addfontfeatures{Scale=1}%
  \fontsize{#1}{#2}\selectfont
  #3%
}

\newcommand{\pitaka}[1]{%
  \par
  \Needspace{4\baselineskip}%
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  % Source Vinayapiṭaka ≈16 bp, medium weight (not bold).
  % Use \centering (not center env) to avoid trivlist \topsep.
  \csromanheadingfont{16bp}{20bp}{\mdseries\csromanplainarg{#1}\par}%
  \nointerlineskip
  % After-skip −20% (was 33bp).
  \vskip 26.4bp\relax
  \endgroup
  \setcsromanpitaka{#1}%
}

\newcommand{\gambhira}[1]{%
  \par
  \Needspace{5\baselineskip}%
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  % Source Pārājikapāḷi ≈36 bp bold (in-page title, not running head).
  \csromanheadingfont{36bp}{42bp}{\bfseries\csromanplainarg{#1}\par}%
  \nointerlineskip
  % After-skips −20% (were 34bp / 24bp).
  \vskip 27.2bp\relax
  % Source rule ≈33 bp wide (~0.09\textwidth).
  \noindent\makebox[\textwidth]{\rule{0.09\textwidth}{0.5pt}}\par
  \nointerlineskip
  \vskip 19.2bp\relax
  \endgroup
  \setcsromanhead{#1}%
  \pagestyle{csroman}%
  \thispagestyle{plain}%
}

% Extra top pad when a chapter opens a sheet (kaṇḍa start).
% Align title top with gambhīra (ปาราชิกปาฬิ) on the opening page
% (measured y0 ≈ 140.8 with body start ≈ 90.7; 72bp sat ~50bp too low).
\newlength{\csromanchaptertopskip}
\setlength{\csromanchaptertopskip}{22bp}

% After \csromanpage / mid-flow: guarantee recto (odd) without a second
% \clearpage when already on an odd empty page.
\newcommand{\csromanensureoddpage}{%
  \ifodd\value{page}\else
    \hbox{}\newpage
  \fi
}

\newcommand{\csromanchapterheadbody}[1]{%
  \begingroup
  \setlength{\parskip}{0pt}%

  \csromanheadingfont{16bp}{20bp}{\bfseries #1\par}%
  \nointerlineskip
  % After-skip −20% (were 15bp).
  \vskip 12bp\relax
  \endgroup
}

% Inline chapter on an already-open sheet (opening stack: Verañjakaṇḍa).
\newcommand{\chapterhead}[1]{%
  \par

  \csromanchapterheadbody{#1}%

% Chapter that opens a printed sheet: odd/recto, no running head, top pad.
% Always clear to a fresh sheet so \vspace* lands in the gambhīra band
% (ปาราชิกปาฬิ y0 ≈ 140.8), not mid-page or on a mixed path.
\newcommand{\chapterheadpage}[1]{%
  \par
  \clearpage
  \csromanensureoddpage
  \thispagestyle{plain}%
  \vspace*{\csromanchaptertopskip}%
  \csromanchapterheadbody{#1}%
}

% Generic section headers under cha: heading_kind h1…h6.
% Best-practice ladder (textbook): never below body (~12 bp ≈ Sarabun 11pt×1.1);
% use size steps only above the floor; pair equal sizes and differentiate with
% weight / italic / spacing. More space above than below so the head attaches
% to the section it introduces.
%
%   h1  14 bp bold          — sole size step under cha (16 bp)
%   h2  13 bp bold          — pair with h3
%   h3  13 bp medium        — same size as h2, lighter weight
%   h4  12 bp bold          — body-size band
%   h5  12 bp italic        — style contrast (not size)
%   h6  12 bp medium        — quietest; spacing mainly
%
% Legacy aliases: \titlehead → h1, \subhead → h2.
\newcommand{\csromanheaderspec}[1]{%
  \ifcase#1\relax
  \or % 1
    \def\csromanheaderfontsize{14bp}%
    \def\csromanheaderbaselineskip{18bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{8pt}%
    \def\csromanheaderafterskip{5pt}%
    \def\csromanheaderneed{3\baselineskip}%
  \or % 2
    \def\csromanheaderfontsize{13bp}%
    \def\csromanheaderbaselineskip{17bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{7pt}%
    \def\csromanheaderafterskip{4pt}%
    \def\csromanheaderneed{2.75\baselineskip}%
  \or % 3 — same size as h2; weight differs
    \def\csromanheaderfontsize{13bp}%
    \def\csromanheaderbaselineskip{17bp}%
    \def\csromanheaderface{\mdseries\upshape}%
    \def\csromanheaderbeforeskip{5.5pt}%
    \def\csromanheaderafterskip{3.5pt}%
    \def\csromanheaderneed{2.5\baselineskip}%
  \or % 4 — body size, bold
    \def\csromanheaderfontsize{12bp}%
    \def\csromanheaderbaselineskip{16bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{5pt}%
    \def\csromanheaderafterskip{3pt}%
    \def\csromanheaderneed{2.25\baselineskip}%
  \or % 5 — body size, italic (style, not shrinkage)
    \def\csromanheaderfontsize{12bp}%
    \def\csromanheaderbaselineskip{16bp}%
    \def\csromanheaderface{\mdseries\itshape}%
    \def\csromanheaderbeforeskip{4pt}%
    \def\csromanheaderafterskip{2.5pt}%
    \def\csromanheaderneed{2\baselineskip}%
  \or % 6 — body size, medium upright; tightest band
    \def\csromanheaderfontsize{12bp}%
    \def\csromanheaderbaselineskip{16bp}%
    \def\csromanheaderface{\mdseries\upshape}%
    \def\csromanheaderbeforeskip{3.5pt}%
    \def\csromanheaderafterskip{2pt}%
    \def\csromanheaderneed{2\baselineskip}%
  \else
    \def\csromanheaderfontsize{14bp}%
    \def\csromanheaderbaselineskip{18bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{8pt}%
    \def\csromanheaderafterskip{5pt}%
    \def\csromanheaderneed{3\baselineskip}%
  \fi
}

\newcommand{\csromanheader}[2]{%
  % #1 = 1..6, #2 = text
  \csromanheaderspec{#1}%
  \Needspace{\csromanheaderneed}%
  \vspace{\csromanheaderbeforeskip}%
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  \csromanheadingfont{\csromanheaderfontsize}{\csromanheaderbaselineskip}{%
    \csromanheaderface #2\par
  }%
  \endgroup
  \vspace{\csromanheaderafterskip}%
}

% Geometry / sandwich centered labels (source_layout=center): body face,
% centered, prose leading +20%. Consecutive lines share that baselineskip —
% not the airy niṭṭhitaṃ band. Avoid ``center`` env (trivlist \topsep).
% Set \baselineskip after \normalsize so the size command does not wipe it.
\newcommand{\csromancenter}[1]{%
  \par
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  \normalsize\mdseries
  \setlength{\baselineskip}{1.2\baselineskip}%
  \csromanplainarg{#1}\par
  \endgroup
}

% Section closers (niṭṭhitaṃ / samattaṃ / end formulas): a quiet structural
% band — lesser than a heading, stronger than body prose. CS Roman source:
% body size, centered. Major closers (kaṇḍa / large sections) may carry
% ``\textbf`` from extract; keep bold weight but never enlarge beyond
% ``\normalsize`` (source uses ~16bp; we mark emphasis with bold only).
% Vertical air: top 14bp; bottom 12.74bp (= 9.8bp + 30%).
% Optional short rule stays on \nitthitamruled / \csromansectionrule.
% Avoid ``center`` env (trivlist).
\newcommand{\nitthitam}[1]{%
  \par
  \Needspace{2.5\baselineskip}%
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  \nointerlineskip
  \vskip 14bp\relax
  {\normalsize #1\par}%
  \nointerlineskip
  \vskip 12.74bp\relax
  \endgroup
}

% Short centered end-of-section rule (``section_rule``). Width matches the
% source underscore glyph run (~33 bp ≈ 0.09\textwidth). Gap above the rule
% equals one body line (\baselineskip). Bottom air 23.4bp (= 18bp + 30%).
\newcommand{\csromansectionrule}{%
  \par
  \Needspace{2.5\baselineskip}%
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  \nointerlineskip
  \vskip\baselineskip\relax
  \noindent\makebox[\textwidth]{\rule{0.09\textwidth}{0.4pt}}\par
  \nointerlineskip
  \vskip 23.4bp\relax
  \endgroup
}

% Closer + end rule as one band (one line between text and rule, air below).
\newcommand{\nitthitamruled}[1]{%
  \par
  \Needspace{3.75\baselineskip}%
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  \nointerlineskip
  \vskip 14bp\relax
  {\normalsize #1\par}%
  \endgroup
  \csromansectionrule
}

% Uddāna / gāthā: optical center (not fixed \gathaindent).
% Per page, \csromangathameasure sets \csromangathapagewidth to the longest
% gāthā line on that page. Each consecutive run (กลุ่ม) is one
% \csromangathagroup: lines left-aligned in a shared-width box, box centered.
% Layout comes from JSON source_layout (bat_line vs wak_line), not line count.
\newlength{\gathaindent}
\setlength{\gathaindent}{65pt}
\newlength{\csromangathapagewidth}
\setlength{\csromangathapagewidth}{0pt}

% #1 = all printed gāthā lines on the page, joined with \\ (no footnotes).
\newcommand{\csromangathameasure}[1]{%
  \settowidth{\csromangathapagewidth}{%
    \begin{tabular}{@{}l@{}}
    #1
    \end{tabular}%
  }%
}

% #1 = one กลุ่ม (≥1 บท): lines joined with \\ ; between full บท use
% \\[\gathastanzaskip]. 1 บทครึ่ง (3 บาท) uses plain \\ only (no stanza skip).
% Left skip + \vtop: top baseline for line spacing; footnotes stay outer.
% Before/after gap = \tipitakaparskip (same air as between prose paragraphs).
\makeatletter
\newcommand{\csromangathagroup}[1]{%
  \par
  \vspace{\tipitakaparskip}%
  \begingroup
  \setlength{\parskip}{0pt}%
  \noindent
  \dimen@=\dimexpr 0.5\linewidth-0.5\csromangathapagewidth\relax
  \ifdim\dimen@<\z@ \dimen@=\z@ \fi
  \hskip\dimen@
  \vtop{%
    \hsize=\csromangathapagewidth
    \linewidth=\hsize
    \raggedright
    \noindent #1\par
  }%
  \par
  \endgroup
  \vspace{\tipitakaparskip}%
}
\makeatother

% Per-segment interword override (layout.json page_layout segments.word_space).
% #1 = scale vs the current font's interword glue (already includes preamble
% WordSpace). Generator passes target/preamble, e.g. 0.8/1.6.
\newcommand{\csromanwordspacebegin}[1]{%
  \begingroup
  \spaceskip=#1\fontdimen2\font
    plus #1\fontdimen3\font
    minus #1\fontdimen4\font
  \relax
}
\newcommand{\csromanwordspaceend}{\endgroup}

% Volume / page layout from segments.json ``layout`` + ``page_layout``.
% Args: word_space factor (vs font WordSpace), line_space, par_indent,
% par_skip, gatha_stanza_skip, gatha_indent, emergency_stretch.
\newcommand{\csromanlayoutapply}[7]{%
  \spaceskip=#1\fontdimen2\font
    plus #1\fontdimen3\font
    minus #1\fontdimen4\font
  \relax
  \linespread{#2}\selectfont
  \setlength{\tipitakahang}{#3}%
  \setlength{\parindent}{\tipitakahang}%
  \setlength{\tipitakaparskip}{#4 plus 1pt minus 0.5pt}%
  \setlength{\parskip}{\tipitakaparskip}%
  \setlength{\gathastanzaskip}{#5}%
  \setlength{\gathaindent}{#6}%
  \setlength{\emergencystretch}{#7}%
}
