% Running-head separator between joined fields (ปิฎก↔นิกาย↔ปาฬิ;
% cha↔h1↔h2 inside \rightmark). Wider than the within-title ``N. Title''
% gap (\spaceskip 0.33em in \csromanrunhead) so parts read as separate names.
\newcommand{\csromanheadsep}{\nobreak\hspace{1em}}

% Join up to three stored verso fields (macro names, no backslash), skipping
% empty ones and separating the present parts with \csromanheadsep.
% \ifcsempty (etoolbox, loaded via memoir) tests the expansion — \ifx with
% \@empty fails for \newcommand-defined (\long) macros even when empty.
\newif\ifcsromanheadpartfirst
\newcommand{\csromanheadjoin}[3]{%
  \csromanheadpartfirsttrue
  \csromanheadjoinpart{#1}%
  \csromanheadjoinpart{#2}%
  \csromanheadjoinpart{#3}%
}
\newcommand{\csromanheadjoinpart}[1]{%
  \ifx\relax#1\relax
  \else
    \ifcsempty{#1}%
      {}%  skip empty field
      {%  emit present field
        \ifcsromanheadpartfirst
          \csromanheadpartfirstfalse
        \else
          \csromanheadsep
        \fi
        \csname #1\endcsname
      }%
  \fi
}

% Running heads — academic / TeX book practice:
%   recede behind body (smaller, medium weight, never bold);
%   folio and title share one face (same size / weight / shape);
%   even/verso:  page | ปิฎก นิกาย ปาฬิ | (empty)
%   odd/recto:   (empty) | cha h1 h2 (marks) | page
% No footer page numbers; no head rule (CS Roman has none).
% Running head uses footnote size so it recedes from body and matches the
% footnote / folio face (one size family across all marginal apparatus).
\newcommand{\csromanrunheadfont}{%
  \normalfont\footnotesize\mdseries\upshape
}

% Extra top pad when a chapter opens a sheet (kaṇḍa start).
% Align title top with gambhīra (ปาราชิกปาฬิ) on the opening page
% (measured y0 ≈ 140.8 with body start ≈ 90.7; 72bp sat ~50bp too low).
\newlength{\csromanchaptertopskip}
\setlength{\csromanchaptertopskip}{\csromandim{22bp}}

% After \csromanpage / mid-flow: guarantee recto (odd) without a second
% \clearpage when already on an odd empty page.
% The inserted verso is intentionally blank — no running head / folio
% (same plain style as chapter opening sheets). Global for all volumes.
\newcommand{\csromanensureoddpage}{%
  \ifodd\value{page}\else
    \hbox{}%
    \thispagestyle{plain}%
    \newpage
  \fi
}

% Blank verso from \cleardoublepage (e.g. after มาติกา → arabic p.1 recto):
% likewise suppress running heads. memoir default uses ``empty''; our
% ``plain'' matches chapter / ensure-odd blanks.
\makeatletter
\renewcommand{\cleardoublepage}{%
  \clearpage
  \if@twoside
    \ifodd\c@page
    \else
      \hbox{}%
      \thispagestyle{plain}%
      \newpage
      \if@twocolumn
        \hbox{}%
        \thispagestyle{plain}%
        \newpage
      \fi
    \fi
  \fi
}
\makeatother

% Chapter that opens a printed sheet: plain running head, top pad.
% After \csromanpage the sheet is already fresh — do not \clearpage again
% (that would bump the sync folio by +1) and do not force recto (odd), which
% would also desync source page numbers.  Only leave the current page when it
% already holds material.
\newcommand{\chapterheadpage}[1]{%
  \par
  \ifdim\pagetotal>0pt
    \clearpage
  \fi
  \thispagestyle{plain}%
  \csromanmarkcha{#1}%
  \vspace*{\csromanchaptertopskip}%
  \csromanchapterheadbody{#1}%
}

% Reading-mode variant: prefer recto when mid-flow (physical pages, not sync).

    \clearpage
  \fi
  \csromanensureoddpage
  \thispagestyle{plain}%
  \csromanmarkcha{#1}%
  \vspace*{\csromanchaptertopskip}%
  \csromanchapterheadbody{#1}%
}

% Generic section headers under cha: heading_kind h1…h6.
% Best-practice ladder (textbook): never below body (~12 bp ≈ Sarabun 11pt×1.1);
% use size steps only above the floor; pair equal sizes and differentiate with
% weight / italic / spacing. More space above than below so the head attaches
% to the section it introduces.
%
%   h1  14 bp bold          — sole size step under cha (16 bp)
%   h2  13 bp bold          — pair with h3
%   h3  13 bp medium        — same size as h2, lighter weight
%   h4  12 bp bold          — body-size band
%   h5  12 bp italic        — style contrast (not size)
%   h6  12 bp medium        — quietest; spacing mainly
%
% Before-skip base tuned for section breaks (h1…h6: 10 / 9 / 7 / 6.5 / 5 / 4.5 pt);
% \csromanheader adds one body \baselineskip on top of that base.
% Legacy aliases: \titlehead → h1, \subhead → h2.
\newcommand{\csromanheaderspec}[1]{%
  \ifcase#1\relax
  \or % 1
    \def\csromanheaderfontsize{14bp}%
    \def\csromanheaderbaselineskip{18bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{10pt}}%
    \def\csromanheaderafterskip{\csromandim{5pt}}%
    \def\csromanheaderneed{3\baselineskip}%
  \or % 2
    \def\csromanheaderfontsize{13bp}%
    \def\csromanheaderbaselineskip{17bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{9pt}}%
    \def\csromanheaderafterskip{\csromandim{4pt}}%
    \def\csromanheaderneed{2.75\baselineskip}%
  \or % 3 — same size as h2; weight differs
    \def\csromanheaderfontsize{13bp}%
    \def\csromanheaderbaselineskip{17bp}%
    \def\csromanheaderface{\mdseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{7pt}}%
    \def\csromanheaderafterskip{\csromandim{3.5pt}}%
    \def\csromanheaderneed{2.5\baselineskip}%
  \or % 4 — body size, bold
    \def\csromanheaderfontsize{12bp}%
    \def\csromanheaderbaselineskip{16bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{6.5pt}}%
    \def\csromanheaderafterskip{\csromandim{3pt}}%
    \def\csromanheaderneed{2.25\baselineskip}%
  \or % 5 — body size, italic (style, not shrinkage)
    \def\csromanheaderfontsize{12bp}%
    \def\csromanheaderbaselineskip{16bp}%
    \def\csromanheaderface{\mdseries\itshape}%
    \def\csromanheaderbeforeskip{\csromandim{5pt}}%
    \def\csromanheaderafterskip{\csromandim{2.5pt}}%
    \def\csromanheaderneed{2\baselineskip}%
  \or % 6 — body size, medium upright; tightest band
    \def\csromanheaderfontsize{12bp}%
    \def\csromanheaderbaselineskip{16bp}%
    \def\csromanheaderface{\mdseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{4.5pt}}%
    \def\csromanheaderafterskip{\csromandim{2pt}}%
    \def\csromanheaderneed{2\baselineskip}%
  \else
    \def\csromanheaderfontsize{14bp}%
    \def\csromanheaderbaselineskip{18bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{10pt}}%
    \def\csromanheaderafterskip{\csromandim{5pt}}%
    \def\csromanheaderneed{3\baselineskip}%
  \fi
}

\newcommand{\csromanheader}[2]{%
  % #1 = 1..6, #2 = text
  % Extra \baselineskip above each level keeps the relative h1…h6 ladder
  % while giving every header roughly one body line of air before it.
  % Recto marks: h1 → clear h2; h2 → keep h1 and set h2; h3…h6 untouched.
  \par
  \ifnum#1=1\relax
    \csromanmarkh{#2}%
  \else
    \ifnum#1=2\relax
      \csromanmarkhii{#2}%
    \fi
  \fi
  \csromanheaderbody{#1}{#2}%
}

% Explicit-page TOC row (sync / deferred outline order).

\newcommand{\csromantocmarkat}[4]{%
  \addtocontents{toc}{\protect\contentsline{#1}{#2}{#3}{#4}}%

% Dest-based TOC row (reading / deferred).  Page resolves via \pageref*.

\newcommand{\csromantocmarkref}[3]{%
  \addtocontents{toc}{%
    \protect\contentsline{#1}{#2}{\protect\pageref*{#3}}{#3}%
  }%

  \fi
}

% Geometry / sandwich centered labels (source_layout=center): body face,

% Section closers (niṭṭhitaṃ / samattaṃ / end formulas): a quiet structural
% band — lesser than a heading, stronger than body prose. CS Roman source:
% body size, centered. Major closers (kaṇḍa / large sections) may carry
% ``\textbf`` from extract; keep bold weight but never enlarge beyond
% ``\normalsize`` (source uses ~16bp; we mark emphasis with bold only).
% Vertical air: top 14bp; bottom 12.74bp (= 9.8bp + 30%).
% Optional short rule stays on \nitthitamruled / \csromansectionrule.
% Avoid ``center`` env (trivlist).
% No \Needspace here: that moved only the closer to the next page when the
% band did not fit, leaving นิฏฺฐิตํ alone at page head. \par\nobreak keeps
% the closer with the preceding paragraph's last lines instead.
\makeatletter
\newcommand{\csroman@closerband}[1]{%
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  \nointerlineskip
  \vskip\csromandim{14bp}\relax
  {\normalsize #1\par}%
  \nointerlineskip
  \vskip\csromandim{12.74bp}\relax
  \endgroup
}
\newcommand{\csroman@closerbandruled}[1]{%
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  \nointerlineskip
  \vskip\csromandim{14bp}\relax
  {\normalsize #1\par}%
  \nointerlineskip
  \nobreak

% #1 = one กลุ่ม of ``\csromangathastanza{...}`` units (generator emits one
% stanza macro per บท; 1 บทครึ่ง is merged into a single stanza).
% Page breaks are allowed *between* stanzas; each stanza is an unbreakable
% \vtop (ไม่แตกบท). Optical center uses \csromangathapagewidth.
% Before/after gap = \tipitakaparskip (same air as between prose paragraphs).
\newlength{\csromangathaleftskip}
\newif\ifcsromangathafirststanza
\makeatletter
\newcommand{\csromangathastanza}[1]{%
  \par
  \ifcsromangathafirststanza
    \csromangathafirststanzafalse
  \else
    % Multi-line \vtop depth breaks normal \baselineskip packing.
    % \gathastanzaskip is the *target baseline-to-baseline* gap between บท
    % (edition default 30pt ≈ body leading + clear stanza air; matches
    % 01Vin01 printing reference). Compensate ~0.62\baselineskip of line-box
    % extent so the measured gap lands near that target. Floor the vskip so
    % a too-small layout value cannot pull บท on top of each other.
    \dimen@=\dimexpr\gathastanzaskip-0.62\baselineskip\relax
    \@tempdima=\dimexpr1.15\baselineskip-0.62\baselineskip\relax
    \ifdim\dimen@<\@tempdima \dimen@=\@tempdima \fi
    \nointerlineskip
    \vskip\dimen@
  \fi
  \noindent
  \hskip\csromangathaleftskip
  \vtop{%
    \hsize=\csromangathapagewidth
    \linewidth=\hsize
    \raggedright
    \noindent #1\par
  }%
  \par
}

% Volume / page layout from segments.json ``layout`` + ``page_layout``.
% Args: word_space factor (vs font WordSpace), line_space, par_indent,

% Printing: scale length args (#3–#7) by \csromanprintingscale; #1/#2 stay unitless.
\newcommand{\csromanlayoutapply}[7]{%
  \spaceskip=#1\fontdimen2\font
    plus #1\fontdimen3\font
    minus #1\fontdimen4\font
  \relax

  \setlength{\tipitakahang}{\dimexpr(#3)*\csromanprintingscalenum/\csromanprintingscaleden\relax}%
  \setlength{\parindent}{\tipitakahang}%
  \setlength{\tipitakaparskip}{\dimexpr(#4)*\csromanprintingscalenum/\csromanprintingscaleden\relax plus 1pt minus 0.5pt}%
  \setlength{\parskip}{\tipitakaparskip}%
  \setlength{\gathastanzaskip}{\dimexpr(#5)*\csromanprintingscalenum/\csromanprintingscaleden\relax}%
  \setlength{\gathaindent}{\dimexpr(#6)*\csromanprintingscalenum/\csromanprintingscaleden\relax}%
  \setlength{\emergencystretch}{\dimexpr(#7)*\csromanprintingscalenum/\csromanprintingscaleden\relax}%
}
