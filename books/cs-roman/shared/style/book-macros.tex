% Closer-chain state: consecutive \\nitthitam / closer bands share the
% optical paragraph gap between lines; bottom air is deferred until the
% next non-closer block (see \\csroman@closerband).
% Under \\nointerlineskip, bare \\tipitakaparskip is shorter than TeX prose
% paragraph air (\\baselineskip packing + \\parskip). Match gāthā stanza
% compensation: \\tipitakaparskip + 0.3\\baselineskip (= optical par_skip).
% Symmetric band: top = bottom = that gap. Deferred bottom is the sole gap
% to the next block — \\csromanapplyclosergap suppresses a stacked
% \\parskip; structural titles use \\addvspace so closer bottom merges with
% their before-skip (max, not sum).
\makeatletter
\newif\ifcsroman@aftercloser
\newif\ifcsroman@noparskip
\newif\ifcsroman@afterheader
\newlength{\csroman@closerbottom}
\newlength{\csroman@prevheaderafter}
% End a structural-title stack so the next h1…h6 / cha uses the full
% before-skip (section break after prose), not the tight inter-title band.
\newcommand{\csromanclearheaderstack}{%
  \global\csroman@afterheaderfalse
  \global\csroman@prevheaderafter\z@
}
\newcommand{\csromanmarkheaderstack}{%
  \global\csroman@afterheadertrue
}
% Record this title's after-skip so a following stacked title can replace it
% (marks / penalties after the glue make \\lastskip unreliable).
\newcommand{\csromanrememberheaderafter}[1]{%
  \global\csroman@prevheaderafter=#1\relax
}
% Choose before-skip: stacked titles → exact \csromanheaderstackskip
% (remove the previous title's after-skip first); else level base.
% Requires \csromanheaderspec already run (defines \csromanheaderbeforeskip).
\newcommand{\csromanapplyheaderbeforeskip}{%
  \ifcsroman@afterheader
    \nointerlineskip
    \ifdim\csroman@prevheaderafter>\z@
      \vskip-\csroman@prevheaderafter
    \fi
    \vskip\csromanheaderstackskip\relax
  \else
    \addvspace{\csromanheaderbeforeskip}%
  \fi
}
\newcommand{\csroman@closergap}{%
  \dimexpr\tipitakaparskip+0.3\baselineskip\relax
}
\newcommand{\csromanflushcloserbottom}{%
  \ifcsroman@aftercloser
    \nointerlineskip
    \addvspace{\csroman@closerbottom}%
    \global\csroman@aftercloserfalse
    \global\csroman@closerbottom\z@
    \global\csroman@prevclosertier\z@
    \global\csroman@noparskiptrue
  \fi
}
% Zero \\parskip for the next paragraph only (call inside a group that
% wraps that paragraph). Clears the flag either way.
\newcommand{\csromanapplyclosergap}{%
  \ifcsroman@noparskip
    \global\csroman@noparskipfalse
    \setlength{\parskip}{0pt}%
  \fi
}
% Structural / already-zero-parskip blocks: consume the flag so it cannot
% leak into the following prose paragraph.
\newcommand{\csromanclearclosergap}{%
  \global\csroman@noparskipfalse
}
\makeatother

% Running-head separator between joined fields (ปิฎก↔นิกาย↔ปาฬิ;

% Join up to three stored verso fields (macro names, no backslash), skipping
% empty ones and separating the present parts with \csromanheadsep.
% \ifcsempty (etoolbox, loaded via memoir) tests the expansion — \ifx with
% \@empty fails for \newcommand-defined (\long) macros even when empty.
\newif\ifcsromanheadpartfirst
\newcommand{\csromanheadjoin}[3]{%
  \csromanheadpartfirsttrue
  \csromanheadjoinpart{#1}%
  \csromanheadjoinpart{#2}%
  \csromanheadjoinpart{#3}%
}
\newcommand{\csromanheadjoinpart}[1]{%
  \ifx\relax#1\relax
  \else
    \ifcsempty{#1}%
      {}%  skip empty field
      {%  emit present field
        \ifcsromanheadpartfirst
          \csromanheadpartfirstfalse
        \else
          \csromanheadsep
        \fi
        \csname #1\endcsname
      }%
  \fi
}

% Mid-volume new pāḷi open (printing/reading): finish the current sheet,
% then guarantee a recto (odd) page so the pāḷi title stack never starts on
% a verso. Blank verso (if inserted) is plain — no running head. First pāḷi
% after มาติกา already sits on a recto via \cleardoublepage; generate emits
% this only before a *subsequent* distinct gambhīra (and its nikāya prelude).
\newcommand{\csromanpalirecto}{%
  \csromanflushcloserbottom
  \par
  \ifdim\pagetotal>0pt
    \clearpage
  \fi
  \csromanensureoddpage
}

\newcommand{\gambhira}[1]{%
  \csromanflushcloserbottom
  \csromanclearclosergap
  \par
  \Needspace{5\baselineskip}%
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  % Book / pāḷi display (``boo``): bold, clearly above ``cha`` (16 bp) but
  % smaller than source Pārājikapāḷi ≈36 bp — long Abhidhamma titles
  % (e.g. ธมฺมานุโลมปจฺจนีย ทุกปฏฺฐานปาฬิ) need room without losing rank.
  \csromanheadingfont{24bp}{30bp}{\bfseries\csromanplainarg{#1}\par}%
  \nointerlineskip
  % After-skips tuned with the 24 bp display (were 27.2bp / 19.2bp at 36 bp).
  \vskip\csromandim{22bp}\relax
  % Source rule ≈33 bp wide (~0.09\textwidth).
  \noindent\makebox[\textwidth]{\rule{0.09\textwidth}{0.5pt}}\par
  \nointerlineskip
  \vskip\csromandim{16bp}\relax
  \endgroup
  \setcsromanheadpali{#1}%
  \pagestyle{csroman}%
  \thispagestyle{plain}%
}

% Extra top pad when a chapter opens a sheet (kaṇḍa start).
% Align title top with gambhīra (ปาราชิกปาฬิ) on the opening page

\newlength{\csromanchaptertopskip}
\setlength{\csromanchaptertopskip}{\csromandim{22bp}}

% After \csromanpage / mid-flow: guarantee recto (odd) without a second
% \clearpage when already on an odd empty page.
% The inserted verso is intentionally blank — no running head / folio
% (same plain style as chapter opening sheets). Global for all volumes.
\newcommand{\csromanensureoddpage}{%
  \ifodd\value{page}\else
    \hbox{}%
    \thispagestyle{plain}%
    \newpage
  \fi
}

% Blank verso from \cleardoublepage (e.g. after มาติกา → arabic p.1 recto):
% likewise suppress running heads. memoir default uses ``empty''; our
% ``plain'' matches chapter / ensure-odd blanks.
\makeatletter
\renewcommand{\cleardoublepage}{%
  \clearpage
  \if@twoside
    \ifodd\c@page
    \else
      \hbox{}%
      \thispagestyle{plain}%
      \newpage
      \if@twocolumn
        \hbox{}%
        \thispagestyle{plain}%
        \newpage
      \fi
    \fi
  \fi
}
\makeatother

% Chapter that opens a printed sheet: plain running head, top pad.
% After \csromanpage the sheet is already fresh — do not \clearpage again
% (that would bump the sync folio by +1) and do not force recto (odd), which
% would also desync source page numbers.  Only leave the current page when it
% already holds material.
\newcommand{\chapterheadpage}[1]{%
  \csromanflushcloserbottom
  \par
  \ifdim\pagetotal>0pt
    \clearpage
  \fi
  \thispagestyle{plain}%
  \csromanmarkcha{#1}%
  \vspace*{\csromanchaptertopskip}%
  \csromanchapterheadbody{#1}%
}

  \fi

  \csromanmarkcha{#1}%
  \vspace*{\csromanchaptertopskip}%
  \csromanchapterheadbody{#1}%
}

% Generic section headers under cha: heading_kind h1…h6.
% Best-practice ladder (textbook): never below body (~12 bp ≈ Sarabun 11pt×1.1);
% use size steps only above the floor; pair equal sizes and differentiate with
% weight / italic / spacing. More space above than below so the head attaches
% to the section it introduces.
%
%   h1  14 bp bold          — sole size step under cha (16 bp)
%   h2  13 bp bold          — pair with h3
%   h3  13 bp medium        — same size as h2, lighter weight
%   h4  12 bp bold          — body-size band
%   h5  12 bp italic        — style contrast (not size)
%   h6  12 bp medium        — quietest; spacing mainly
%
% Before-skip (after prose / section break): h1…h6 bases
% 10 / 9 / 7 / 6.5 / 5 / 4.5 pt — more above than below; no extra
% \baselineskip (that doubled air in title stacks).
% Stacked titles (cha/h1…h6 immediately after another): use
% \csromanheaderstackskip only (~source saṃyutta→vagga ink gap ≈17pt).
% \csromanheaderneed ≈ vertical size of the heading band (before+title+after);
% \csromanheaderbody adds +3\baselineskip so ≥2 body lines must fit with it.
% Legacy aliases: \titlehead → h1, \subhead → h2.
\newcommand{\csromanheaderstackskip}{\csromandim{14pt}}
\newcommand{\csromanheaderspec}[1]{%
  \ifcase#1\relax
  \or % 1
    \def\csromanheaderfontsize{14bp}%
    \def\csromanheaderbaselineskip{18bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{10pt}}%
    \def\csromanheaderafterskip{\csromandim{5pt}}%
    \def\csromanheaderneed{3\baselineskip}%
  \or % 2
    \def\csromanheaderfontsize{13bp}%
    \def\csromanheaderbaselineskip{17bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{9pt}}%
    \def\csromanheaderafterskip{\csromandim{4pt}}%
    \def\csromanheaderneed{2.75\baselineskip}%
  \or % 3 — same size as h2; weight differs
    \def\csromanheaderfontsize{13bp}%
    \def\csromanheaderbaselineskip{17bp}%
    \def\csromanheaderface{\mdseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{7pt}}%
    \def\csromanheaderafterskip{\csromandim{3.5pt}}%
    \def\csromanheaderneed{2.5\baselineskip}%
  \or % 4 — body size, bold
    \def\csromanheaderfontsize{12bp}%
    \def\csromanheaderbaselineskip{16bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{6.5pt}}%
    \def\csromanheaderafterskip{\csromandim{3pt}}%
    \def\csromanheaderneed{2.25\baselineskip}%
  \or % 5 — body size, italic (style, not shrinkage)
    \def\csromanheaderfontsize{12bp}%
    \def\csromanheaderbaselineskip{16bp}%
    \def\csromanheaderface{\mdseries\itshape}%
    \def\csromanheaderbeforeskip{\csromandim{5pt}}%
    \def\csromanheaderafterskip{\csromandim{2.5pt}}%
    \def\csromanheaderneed{2\baselineskip}%
  \or % 6 — body size, medium upright; tightest band
    \def\csromanheaderfontsize{12bp}%
    \def\csromanheaderbaselineskip{16bp}%
    \def\csromanheaderface{\mdseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{4.5pt}}%
    \def\csromanheaderafterskip{\csromandim{2pt}}%
    \def\csromanheaderneed{2\baselineskip}%
  \else
    \def\csromanheaderfontsize{14bp}%
    \def\csromanheaderbaselineskip{18bp}%
    \def\csromanheaderface{\bfseries\upshape}%
    \def\csromanheaderbeforeskip{\csromandim{10pt}}%
    \def\csromanheaderafterskip{\csromandim{5pt}}%
    \def\csromanheaderneed{3\baselineskip}%
  \fi
}

\newcommand{\csromanheader}[2]{%
  % #1 = 1..6, #2 = text
  % Recto marks: h1 → clear h2; h2 → keep h1 and set h2; h3…h6 untouched.
  \csromanflushcloserbottom
  \par
  \ifnum#1=1\relax
    \csromanmarkh{#2}%
  \else
    \ifnum#1=2\relax
      \csromanmarkhii{#2}%
    \fi
  \fi
  \csromanheaderbody{#1}{#2}%
}

\newcommand{\csromanheaderbody}[2]{%
  % #1 = 1..6, #2 = text
  % Keep-with-next: Needspace alone is not enough — if only the title band
  % fits, TeX still breaks before the following prose (orphan heading at page

  % Flush deferred closer bottom *before* Needspace so it is not left between
  % the title and the following prose. \\addvspace merges closer bottom /
  % previous title after-skip with this before-skip (max, not sum).
  % \csromanapplyheaderbeforeskip: stack skip when previous block was a title.
  \csromanflushcloserbottom
  \csromanclearclosergap
  \csromanheaderspec{#1}%
  \Needspace{\dimexpr\csromanheaderneed+3\baselineskip\relax}%
  \csromanapplyheaderbeforeskip
  \begingroup
  \setlength{\parskip}{0pt}%
  \centering
  \csromanheadingfont{\csromanheaderfontsize}{\csromanheaderbaselineskip}{%
    \csromanheaderface #2\par
  }%
  \nointerlineskip
  \endgroup
  \nobreak
  \addvspace{\csromanheaderafterskip}%
  \csromanrememberheaderafter{\csromanheaderafterskip}%
  \nobreak
  \csromanmarkheaderstack
}

% restriction as \\marginpar → use \\marginnote for folio). Defer note
% bodies while building the box; flush after it joins the main list
% (\\csromanfn@defer* in footnotes.tex).
\newlength{\csromangathaleftskip}
\newif\ifcsromangathafirststanza
\makeatletter
\newcommand{\csromangathastanza}[1]{%
  \par
  \ifcsromangathafirststanza
    \csromangathafirststanzafalse
  \else
    % Prose inter-paragraph air is NOT \parskip alone: TeX adds \parskip on
    % top of normal \baselineskip packing. A multi-line \vtop breaks that
    % packing (its depth spans every line), so emulate the optical gap with
    % \nointerlineskip plus an explicit skip:
    %   \gathastanzaskip  (= layout gatha_stanza_skip, keep equal to par_skip)
    %   + 0.3\baselineskip (open leading remainder; tuned so ink-to-ink gap
    %     on 01Vin01 printing matches prose at par_skip=gatha_stanza_skip=8pt)
    \nointerlineskip
    \vskip\dimexpr\gathastanzaskip+0.3\baselineskip\relax
  \fi
  \global\csromanfn@defertrue
  \noindent
  \hskip\csromangathaleftskip
  \vtop{%
    \hsize=\csromangathapagewidth
    \linewidth=\hsize
    \raggedright
    \noindent #1\par
  }%
  \csromanfn@flushdeferred
  \par
}
\newcommand{\csromangathagroup}[1]{%
  \csromanflushcloserbottom
  \csromanclearclosergap
  \csromanclearheaderstack
  \par
  \addvspace{\tipitakaparskip}%
  \begingroup
  \setlength{\parskip}{0pt}%
  \dimen@=\dimexpr 0.5\linewidth-0.5\csromangathapagewidth\relax
  \ifdim\dimen@<\z@ \dimen@=\z@ \fi
  \setlength{\csromangathaleftskip}{\dimen@}%
  \csromangathafirststanzatrue
  #1%
  \endgroup
  \addvspace{\tipitakaparskip}%
}

% Interword glue = factor × current font's fontdimen 2/3/4 (includes preamble
% WordSpace). Re-call after any size change (\footnotesize, etc.) so absolute
% body \spaceskip is not inherited by smaller faces.
\makeatletter
\newcommand{\csroman@wordspacefactor}{1}
\newcommand{\csromansetwordspace}[1]{%
  \spaceskip=#1\fontdimen2\font
    plus #1\fontdimen3\font
    minus #1\fontdimen4\font
  \relax
}

% Per-segment interword override (absolute layout word_space vs preamble).
% #1 = scale vs the current font's interword glue (includes preamble WordSpace).
% Generator passes target/preamble_WordSpace, e.g. 0.8/3.5.
% Does not overwrite \csroman@wordspacefactor (volume / page layout owns that).
% Footnotes clear \spaceskip after \footnotesize (font natural glue), not this factor.
\newcommand{\csromanwordspacebegin}[1]{%
  \begingroup
  \csromansetwordspace{#1}%
}
\newcommand{\csromanwordspaceend}{\endgroup}

% Volume / page layout from segments.json ``layout`` + ``page_layout``.
% Args: word_space factor (vs font WordSpace), line_space, par_indent,
% par_skip, gatha_stanza_skip, gatha_indent, emergency_stretch.
% Printing: scale length args (#3–#7) by \csromanprintingscale; #1/#2 stay unitless.
\newcommand{\csromanlayoutapply}[7]{%
  \renewcommand{\csroman@wordspacefactor}{#1}%
  \csromansetwordspace{#1}%
  \linespread{#2}\selectfont
  \setlength{\tipitakahang}{\dimexpr(#3)*\csromanprintingscalenum/\csromanprintingscaleden\relax}%
  \setlength{\parindent}{\tipitakahang}%
  \setlength{\tipitakaparskip}{\dimexpr(#4)*\csromanprintingscalenum/\csromanprintingscaleden\relax plus 1pt minus 0.5pt}%
  \setlength{\parskip}{\tipitakaparskip}%
  \setlength{\gathastanzaskip}{\dimexpr(#5)*\csromanprintingscalenum/\csromanprintingscaleden\relax}%
  \setlength{\gathaindent}{\dimexpr(#6)*\csromanprintingscalenum/\csromanprintingscaleden\relax}%
  \setlength{\emergencystretch}{\dimexpr(#7)*\csromanprintingscalenum/\csromanprintingscaleden\relax}%
}
