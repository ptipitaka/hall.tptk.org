% =========================================================================
% footnotes.tex — adaptive 1/2/3-column footnotes for CS Roman
% =========================================================================
% Goal: save vertical space. Prefer packing 2–3 notes on one line whenever
% their natural one-line widths (+ column seps) fit in \columnwidth.
%
% Natural width = footmisc hang layout
%   (\hb@xt@\footnotemargin{\hss mark}body), not mark+\footnotemargin+body.
% Thai above/below marks (ิ ี ุ ู ํ ฺ …) have zero advance; \settowidth /
% \wd of that hang hbox matches visual width.
%
% Classification (per note, from natural box width):
%   • > \columnwidth → long (must wrap; full-width 1-col row)
%   • > third measure → short (prefer start/join 2-col row)
%   • ≤ third measure → tiny (prefer collect toward 3-col row)
%
% Packing (delivery order; pending cells):
%   • pending cells are the sealed natural boxes (not equal half/third cells)
%   • complete 2/3-col rows: min \csromanfn@colsep between cells, stretch
%     (plus 1fil) so the row is edge-to-edge on \columnwidth
%   • single-item row when the page has only one note and it fits one line:
%     center (\hss + note + \hss)
%   • single-item row when the page has 2+ notes (trailing leftover), or the
%     sole note is long/wrapping: left-align (note + \hss)
%   • join when left[+mid]+new + seps ≤ \columnwidth; else flush and restart
%   • long notes flush pending, then emit the sealed long vbox
% Vertical rhythm: fixed \csromanfn@rowsep between rows.
%
% Page break / mark–body sync:
%   Each \footnote \insert\footins the note's natural hang hbox (or long
%   vbox) with \floatingpenalty\@MM. TeX seals those boxes into \box\footins
%   with their callout lines. \@makecol unpacks exactly those sealed boxes and
%   repacks them into 1/2/3-col rows — no parallel toks queue (FIFO desync
%   painted wrong bodies / bodies before callouts).
% Contract: footnote mark and body share the same physical page.
% Preserves footmisc hang + hyperref patches (AtBeginDocument).
% =========================================================================

\makeatletter

\def\footnoterule{%
  \kern-3\p@
  \hrule\@width\columnwidth\@height.4pt
  \kern10\p@
}

% Footnotes stay pinned to the true page bottom (footmisc `bottom`): this
% keeps footnote rules aligned across facing pages, which matters more
% than a constant body->rule gap. (Tried patching footmisc's bottom-pin
% \vfill to a fixed \vskip — that flattened the gap but broke rule
% alignment across a spread; reverted.)
%
% Packing vs page-break accounting: each note is \insert'ed as its own
% sealed box, so TeX reserves *stacked* height at break time. \@makecol
% then repacks 2–3 notes onto one row, freeing vertical space. Without a
% credit, that surplus becomes body→rule gap under bottom-pin \vfill
% (e.g. 01Vin01 reading p.20: three tiny notes → one row left ~2 body
% lines unused even though the paragraph could continue). Non-long notes
% therefore insert a negative \vskip *before* the sealed box so the page
% builder reserves ~1/2 (short) or ~1/3 (tiny) height — matching the
% packer. Collect's \unskip strips the credit; long notes keep full height.

\newcount\csromanfn@pending
\newcount\csromanfn@mode
\newcount\csromanfn@n
\newcount\csromanfn@i
\newdimen\csromanfn@colsep
\newdimen\csromanfn@colw
\newdimen\csromanfn@halfw
\newdimen\csromanfn@thirdw
\newdimen\csromanfn@nat
\newdimen\csromanfn@leftnat
\newdimen\csromanfn@midnat
\newdimen\csromanfn@sum
\newdimen\csromanfn@rowsep
\newbox\csromanfn@box
\newbox\csromanfn@left
\newbox\csromanfn@mid
\newbox\csromanfn@right
\newbox\csromanfn@tmp
\newif\ifcsromanfn@anyrow
\newif\ifcsromanfn@longitem

% Max notes packed from one page's footins (volume pages stay well below).
\newcount\csromanfn@maxitems
\csromanfn@maxitems=64
\csromanfn@i=\z@
\loop
  \ifnum\csromanfn@i<\csromanfn@maxitems
    \advance\csromanfn@i\@ne
    \expandafter\newbox\csname csromanfn@item@\the\csromanfn@i\endcsname
\repeat

\csromanfn@colsep=1.25em
\csromanfn@rowsep=1.5pt

\def\csromanfn@reset{%
  \global\csromanfn@n=\z@
  \csromanfn@clearpending
}

\def\csromanfn@clearpending{%
  \global\csromanfn@pending=\z@
  \global\csromanfn@mode=\z@
  \global\csromanfn@leftnat=\z@
  \global\csromanfn@midnat=\z@
  \global\setbox\csromanfn@left=\box\voidb@x
  \global\setbox\csromanfn@mid=\box\voidb@x
  \global\setbox\csromanfn@right=\box\voidb@x
}

\def\csromanfn@beforerow{%
  \ifcsromanfn@anyrow
    \vskip\csromanfn@rowsep
  \fi
  \csromanfn@anyrowtrue
}

\def\csromanfn@emitrownaturalone{%
  \csromanfn@beforerow
  \hbox to\columnwidth{%
    % Center only when this page has a single one-line note. Multi-note
    % pages left-align a trailing singleton (01Vin01 p.86 * note).
    \ifnum\csromanfn@n>\@ne
      \box\csromanfn@left
      \hss
    \else
      \hss
      \box\csromanfn@left
      \hss
    \fi
  }%
  \csromanfn@clearpending
}

\def\csromanfn@emitrownaturaltwo{%
  \csromanfn@beforerow
  \hbox to\columnwidth{%
    \box\csromanfn@left
    \hskip\csromanfn@colsep plus 1fil
    \box\csromanfn@right
  }%
  \csromanfn@clearpending
}

\def\csromanfn@emitrownaturalthree{%
  \csromanfn@beforerow
  \hbox to\columnwidth{%
    \box\csromanfn@left
    \hskip\csromanfn@colsep plus 1fil
    \box\csromanfn@mid
    \hskip\csromanfn@colsep plus 1fil
    \box\csromanfn@right
  }%
  \csromanfn@clearpending
}

\def\csromanfn@emitpending{%
  \ifnum\csromanfn@pending=\@ne
    \csromanfn@emitrownaturalone
  \else
    \ifnum\csromanfn@pending=\tw@
      \global\setbox\csromanfn@right=\box\csromanfn@mid
      \csromanfn@emitrownaturaltwo
    \else
      \ifnum\csromanfn@pending>\tw@
        \csromanfn@emitrownaturalthree
      \fi
    \fi
  \fi
}

% Emit a sealed long vbox (full measure).
\def\csromanfn@emitlongbox{%
  \csromanfn@emitpending
  \csromanfn@beforerow
  \box\csromanfn@box
}

% Classify \csromanfn@box: long if vbox or wd > columnwidth.
\def\csromanfn@classbox{%
  \csromanfn@longitemfalse
  \ifvbox\csromanfn@box
    \csromanfn@longitemtrue
  \else
    \ifdim\wd\csromanfn@box>\columnwidth
      \csromanfn@longitemtrue
    \fi
  \fi
}

% Add sealed non-long \csromanfn@box into the row packer.
\def\csromanfn@addbox{%
  \csromanfn@classbox
  \ifcsromanfn@longitem
    \csromanfn@emitlongbox
  \else
    \csromanfn@nat=\wd\csromanfn@box
    \ifnum\csromanfn@pending=\z@
      \global\csromanfn@leftnat=\csromanfn@nat
      \global\setbox\csromanfn@left=\box\csromanfn@box
      \global\csromanfn@pending=\@ne
      \ifdim\csromanfn@nat>\csromanfn@thirdw
        \global\csromanfn@mode=2\relax
      \else
        \global\csromanfn@mode=3\relax
      \fi
    \else
      \ifnum\csromanfn@mode=3\relax
        \ifnum\csromanfn@pending=\@ne
          \ifdim\csromanfn@nat>\csromanfn@thirdw
            \csromanfn@sum=\dimexpr
              \csromanfn@leftnat+\csromanfn@colsep+\csromanfn@nat\relax
            \ifdim\csromanfn@sum>\columnwidth
              \csromanfn@emitpending
              \csromanfn@addbox
            \else
              \global\setbox\csromanfn@right=\box\csromanfn@box
              \csromanfn@emitrownaturaltwo
            \fi
          \else
            \global\csromanfn@midnat=\csromanfn@nat
            \global\setbox\csromanfn@mid=\box\csromanfn@box
            \global\csromanfn@pending=\tw@
          \fi
        \else
          \csromanfn@sum=\dimexpr
            \csromanfn@leftnat+\csromanfn@colsep
            +\csromanfn@midnat+\csromanfn@colsep
            +\csromanfn@nat\relax
          \ifdim\csromanfn@sum>\columnwidth
            \csromanfn@emitpending
            \csromanfn@addbox
          \else
            \global\setbox\csromanfn@right=\box\csromanfn@box
            \csromanfn@emitrownaturalthree
          \fi
        \fi
      \else
        \csromanfn@sum=\dimexpr
          \csromanfn@leftnat+\csromanfn@colsep+\csromanfn@nat\relax
        \ifdim\csromanfn@sum>\columnwidth
          \csromanfn@emitpending
          \csromanfn@addbox
        \else
          \global\setbox\csromanfn@right=\box\csromanfn@box
          \csromanfn@emitrownaturaltwo
        \fi
      \fi
    \fi
  \fi
}

% Unpack sealed insert boxes from #1 into named item boxes; set \csromanfn@n.
\def\csromanfn@collectfrom#1{%
  \global\csromanfn@n=\z@
  \setbox\csromanfn@tmp=\vbox{%
    \unvbox#1%
    \loop
      \unskip
      \unpenalty
      \setbox\csromanfn@box=\lastbox
    \ifvoid\csromanfn@box
    \else
      \global\advance\csromanfn@n\@ne
      \ifnum\csromanfn@n>\csromanfn@maxitems
        \errmessage{csromanfn: too many footnotes on one page}%
      \else
        \global\expandafter\setbox\csname csromanfn@item@\the\csromanfn@n\endcsname
          =\box\csromanfn@box
      \fi
    \repeat
  }%
}

\def\csromanfn@packitems{%
  \csromanfn@clearpending
  \global\csromanfn@anyrowfalse
  % lastbox collect stores last sealed note as item@1 — emit n..1.
  \ifnum\csromanfn@n>\z@
    \csromanfn@i=\csromanfn@n
    \loop
      \ifnum\csromanfn@i>\z@
        \global\setbox\csromanfn@box=
          \expandafter\box\csname csromanfn@item@\the\csromanfn@i\endcsname
        \csromanfn@addbox
        \advance\csromanfn@i\m@ne
    \repeat
    \csromanfn@emitpending
  \fi
}

\def\csromanfn@buildfootins{%
  \ifvoid\footins
  \else
    \setbox\csromanfn@tmp=\box\footins
    \csromanfn@collectfrom{\csromanfn@tmp}%
    \global\setbox\footins=\vbox\bgroup
      \reset@font\footnotesize
      \frenchspacing
      \offinterlineskip
      \floatingpenalty\@MM
      \csromanfn@packitems
    \egroup
  \fi
}

% Build natural note box in \csromanfn@box (hang hbox or long vbox).
\long\def\csromanfn@makenote#1{%
  \settowidth\csromanfn@nat{%
    \reset@font\footnotesize
    \frenchspacing
    \hbox{\hb@xt@\footnotemargin{\hss\@makefnmark}#1}%
  }%
  \ifdim\csromanfn@nat>\columnwidth
    \global\setbox\csromanfn@box=\vbox{%
      \reset@font\footnotesize
      \frenchspacing
      \hsize\columnwidth
      \columnwidth\columnwidth
      \linewidth\columnwidth
      \@parboxrestore
      \protected@edef\@currentlabel{%
        \csname p@footnote\endcsname\@thefnmark
      }%
      \@makefntext{%
        \rule\z@\footnotesep\ignorespaces#1\@finalstrut\strutbox}%
      \par
    }%
  \else
    \global\setbox\csromanfn@box=\hbox{%
      \reset@font\footnotesize
      \frenchspacing
      \hbox{\hb@xt@\footnotemargin{\hss\@makefnmark}#1}%
    }%
  \fi
}

\long\def\csromanfn@footnotetext#1{%
  \csromanfn@makenote{#1}%
  % Net insert size for page-break accounting (see preamble comment above).
  \csromanfn@sum=\dimexpr\ht\csromanfn@box+\dp\csromanfn@box\relax
  \insert\footins{%
    \floatingpenalty\@MM
    \ifdim\csromanfn@nat>\columnwidth
    \else
      \ifdim\csromanfn@nat>\dimexpr\columnwidth/3\relax
        \vskip-.5\csromanfn@sum
      \else
        \vskip-\dimexpr2\csromanfn@sum/3\relax
      \fi
    \fi
    \box\csromanfn@box
  }%
}

\def\csromanfn@makecol{%
  \ifvoid\footins
  \else
    \csromanfn@buildfootins
  \fi
  \csromanfn@saved@makecol
  \csromanfn@clearpending
}

\AtBeginDocument{%
  \csromanfn@halfw=\dimexpr(\columnwidth-\csromanfn@colsep)/2\relax
  \csromanfn@thirdw=\dimexpr(\columnwidth-2\csromanfn@colsep)/3\relax
  \csromanfn@colw=\csromanfn@halfw
  \csromanfn@reset
  \let\csromanfn@saved@footnotetext\@footnotetext
  \let\csromanfn@saved@makecol\@makecol
  \let\@footnotetext\csromanfn@footnotetext
  \let\@makecol\csromanfn@makecol
}

\makeatother
