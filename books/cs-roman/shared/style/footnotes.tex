% =========================================================================
% footnotes.tex — adaptive 1/2/3-column footnotes for CS Roman
% =========================================================================
% Packing rule (per page, walk notes in order):
%   Width thresholds (natural one-line width of mark + body) — all marks
%   (numbered and apparatus * / +) use the same cuts:
%     • > half measure  → long → full-width 1-col row
%     • > third measure → short (2-col candidate)
%     • ≤ third measure → tiny (3-col candidate)
%   If a short/tiny note still wraps (\prevgraf>1) inside its column box,
%   promote (tiny→retry half, else → long). Do not use box height: Thai
%   single-line notes exceed ~1.55\baselineskip via \footnotesep + marks.
%   Display order (CS Roman convention): numbered notes first, then * / +.
%   Short numbered + short apparatus may share one row (do not flush between
%   the two passes — otherwise a lone short ¹ never pairs with * / +).
%   Long notes interrupt the short/tiny row (flush pending cells first).
%   Non-long notes fill a row left-to-right:
%     • start in 3-col mode when the first note is tiny (≤ third)
%     • start in 2-col mode when the first note is short (≤ half only)
%     • a short note cannot join a 3-col row with 2 tinies → emit, restart
%     • a short note + one pending tiny → promote to 2-col and pair
%     • trailing incomplete row emits with empty right/mid cells
%   Vertical rhythm: fixed \csromanfn@rowsep between rows (mark-independent).
% Page break: each note reserves height via \insert\footins (empty vbox).
%   Content lives in \csromanfn@toks. At \@makecol, only notes whose
%   reserved height fits in the current \box\footins are packed; the rest
%   stay in toks for the next page (fixes marks on p.N+1 with text on p.N
%   when a paragraph spans the break).
% Preserves footmisc hang + hyperref patches (AtBeginDocument).
% =========================================================================

\makeatletter

\def\footnoterule{%
  \kern-3\p@
  \hrule\@width\columnwidth\@height.4pt
  \kern3.38\p@
}

\csromanfn@colsep=1.25em
\csromanfn@rowsep=1.5pt

\def\csromanfn@reset{%
  \global\csromanfn@toks{}%
  \global\csromanfn@pagetoks{}%
  \global\csromanfn@nexttoks{}%
  \global\csromanfn@n=\z@
  \global\csromanfn@pagen=\z@
  \global\csromanfn@nextn=\z@
  \csromanfn@clearpending
}

% Clear row-packing state only (keep deferred toks / n across pages).
\def\csromanfn@clearpending{%
  \global\csromanfn@pending=\z@
  \global\csromanfn@mode=\z@
  \global\setbox\csromanfn@left=\box\voidb@x
  \global\setbox\csromanfn@mid=\box\voidb@x
  \global\setbox\csromanfn@right=\box\voidb@x
}

% True when #1 is apparatus mark * or +.
\def\csromanfn@issymbol#1{%
  \def\csromanfn@tmp{#1}%
  \def\csromanfn@star{*}%
  \def\csromanfn@plus{+}%
  \ifx\csromanfn@tmp\csromanfn@star
    \csromanfn@symboltrue
  \else
    \ifx\csromanfn@tmp\csromanfn@plus
      \csromanfn@symboltrue
    \else
      \csromanfn@symbolfalse
    \fi
  \fi
}

% #1=mark #2=body #3=class: 0 tiny / 1 short / 2 long  #4=reserved height
\long\def\csromanfn@append#1#2#3#4{%
  \toks@=\expandafter{\the\csromanfn@toks\csromanfn@do{#1}{#2}{#3}{#4}}%
  \global\csromanfn@toks=\expandafter{\the\toks@}%
}

\def\csromanfn@beforerow{%
  \ifcsromanfn@anyrow
    \vskip\csromanfn@rowsep
  \fi
  \csromanfn@anyrowtrue
}

\long\def\csromanfn@typesetone#1#2{%
  \def\@thefnmark{#1}%
  \@makefntext{%
    \rule\z@\footnotesep\ignorespaces#2\@finalstrut\strutbox}%
  \par
}

\long\def\csromanfn@typesetlong#1#2{%
  \csromanfn@beforerow
  \vtop{%
    \reset@font\footnotesize
    \frenchspacing
    \hsize\columnwidth
    \columnwidth\columnwidth
    \linewidth\columnwidth
    \@parboxrestore
    \leftskip\z@\@plus\z@\rightskip\z@\@plus\z@
    \csromanfn@typesetone{#1}{#2}%
  }%
}

% Wrap = more than one paragraph line (\prevgraf), not box height.
% Height checks false-fire on Thai: \footnotesep strut + upper/lower marks
% make a single line taller than ~1.55\baselineskip.
\long\def\csromanfn@makenotebox#1#2{%
  \global\setbox\csromanfn@box=\vtop{%
    \reset@font\footnotesize
    \frenchspacing
    \hsize\csromanfn@colw
    \columnwidth\csromanfn@colw
    \linewidth\csromanfn@colw
    \@parboxrestore
    \leftskip\z@\@plus\z@\rightskip\z@\@plus\z@
    \csromanfn@typesetone{#1}{#2}%
    \global\csromanfn@linecount=\prevgraf
  }%
  \ifnum\csromanfn@linecount>\@ne
    \csromanfn@wrappedtrue
  \else
    \csromanfn@wrappedfalse
  \fi
}

\def\csromanfn@emitrowtwo{%
  \csromanfn@beforerow
  \hbox to\columnwidth{%
    \hbox to\csromanfn@halfw{\box\csromanfn@left\hss}%
    \hskip\csromanfn@colsep
    \hbox to\csromanfn@halfw{%
      \ifvoid\csromanfn@right \else \box\csromanfn@right \fi
      \hss
    }%
  }%
  \global\setbox\csromanfn@left=\box\voidb@x
  \global\setbox\csromanfn@right=\box\voidb@x
  \global\csromanfn@pending=\z@
  \global\csromanfn@mode=\z@
}

\def\csromanfn@emitrowthree{%
  \csromanfn@beforerow
  \hbox to\columnwidth{%
    \hbox to\csromanfn@thirdw{\box\csromanfn@left\hss}%
    \hskip\csromanfn@colsep
    \hbox to\csromanfn@thirdw{%
      \ifvoid\csromanfn@mid \else \box\csromanfn@mid \fi
      \hss
    }%
    \hskip\csromanfn@colsep
    \hbox to\csromanfn@thirdw{%
      \ifvoid\csromanfn@right \else \box\csromanfn@right \fi
      \hss
    }%
  }%
  \global\setbox\csromanfn@left=\box\voidb@x
  \global\setbox\csromanfn@mid=\box\voidb@x
  \global\setbox\csromanfn@right=\box\voidb@x
  \global\csromanfn@pending=\z@
  \global\csromanfn@mode=\z@
}

\def\csromanfn@emitpending{%
  \ifnum\csromanfn@pending>\z@
    \ifnum\csromanfn@mode=3\relax
      \csromanfn@emitrowthree
    \else
      \csromanfn@emitrowtwo
    \fi
  \fi
}

% Note wrapped in its column → flush pending peers, then full-width long.
\long\def\csromanfn@promotelong#1#2{%
  \csromanfn@emitpending
  \csromanfn@typesetlong{#1}{#2}%
}

% Add one non-long note into the current row (or start a new row).
\long\def\csromanfn@addshort#1#2#3{%
  \ifnum\csromanfn@pending=\z@
    \ifnum#3=\z@
      \global\csromanfn@mode=3\relax
      \csromanfn@colw=\csromanfn@thirdw
    \else
      \global\csromanfn@mode=2\relax
      \csromanfn@colw=\csromanfn@halfw
    \fi
    \csromanfn@makenotebox{#1}{#2}%
    \ifcsromanfn@wrapped
      % Tiny wrapped in a third-col → retry as short (half) before 1-col.
      \ifnum#3=\z@
        \global\csromanfn@mode=2\relax
        \csromanfn@colw=\csromanfn@halfw
        \csromanfn@makenotebox{#1}{#2}%
        \ifcsromanfn@wrapped
          \global\csromanfn@mode=\z@
          \csromanfn@typesetlong{#1}{#2}%
        \else
          \global\setbox\csromanfn@left=\box\csromanfn@box
          \global\csromanfn@pending=\@ne
        \fi
      \else
        \global\csromanfn@mode=\z@
        \csromanfn@typesetlong{#1}{#2}%
      \fi
    \else
      \global\setbox\csromanfn@left=\box\csromanfn@box
      \global\csromanfn@pending=\@ne
    \fi
  \else
    \ifnum\csromanfn@mode=3\relax
      \ifnum#3>\z@
        % Short cannot sit in a 3-col row. If only one tiny is pending,
        % promote to 2-col and pair (e.g. short ¹ + apparatus *).
        \ifnum\csromanfn@pending=\@ne
          \global\csromanfn@mode=2\relax
          \csromanfn@colw=\csromanfn@halfw
          \csromanfn@makenotebox{#1}{#2}%
          \ifcsromanfn@wrapped
            \csromanfn@promotelong{#1}{#2}%
          \else
            \global\setbox\csromanfn@right=\box\csromanfn@box
            \csromanfn@emitrowtwo
          \fi
        \else
          \csromanfn@emitpending
          \csromanfn@addshort{#1}{#2}{#3}%
        \fi
      \else
        \csromanfn@colw=\csromanfn@thirdw
        \csromanfn@makenotebox{#1}{#2}%
        \ifcsromanfn@wrapped
          \csromanfn@promotelong{#1}{#2}%
        \else
          \ifnum\csromanfn@pending=\@ne
            \global\setbox\csromanfn@mid=\box\csromanfn@box
            \global\csromanfn@pending=\tw@
          \else
            \global\setbox\csromanfn@right=\box\csromanfn@box
            \csromanfn@emitrowthree
          \fi
        \fi
      \fi
    \else
      % 2-col mode: this note completes the pair.
      \csromanfn@colw=\csromanfn@halfw
      \csromanfn@makenotebox{#1}{#2}%
      \ifcsromanfn@wrapped
        \csromanfn@promotelong{#1}{#2}%
      \else
        \global\setbox\csromanfn@right=\box\csromanfn@box
        \csromanfn@emitrowtwo
      \fi
    \fi
  \fi
}

\long\def\csromanfn@walkone#1#2#3#4{%
  \ifnum#3>1\relax
    \csromanfn@emitpending
    \csromanfn@typesetlong{#1}{#2}%
  \else
    \csromanfn@addshort{#1}{#2}{#3}%
  \fi
}

% Two-pass walk: numbered first, then apparatus (* / +).
\long\def\csromanfn@walknumbered#1#2#3#4{%
  \csromanfn@issymbol{#1}%
  \ifcsromanfn@symbol\else
    \csromanfn@walkone{#1}{#2}{#3}{#4}%
  \fi
}

\long\def\csromanfn@walksymbol#1#2#3#4{%
  \csromanfn@issymbol{#1}%
  \ifcsromanfn@symbol
    \csromanfn@walkone{#1}{#2}{#3}{#4}%
  \fi
}

% Split toks into this-page vs deferred using reserved heights vs \box\footins.
\long\def\csromanfn@partition#1#2#3#4{%
  \ifcsromanfn@holding
    \toks@=\expandafter{\the\csromanfn@nexttoks\csromanfn@do{#1}{#2}{#3}{#4}}%
    \global\csromanfn@nexttoks=\expandafter{\the\toks@}%
    \global\advance\csromanfn@nextn\@ne
  \else
    \csromanfn@ht=\dimexpr\csromanfn@htsum+#4\relax
    % 0.5pt slack for rounding when comparing placeholder heights.
    \ifdim\csromanfn@ht>\dimexpr\csromanfn@htheld+0.5pt\relax
      \csromanfn@holdingtrue
      \toks@=\expandafter{\the\csromanfn@nexttoks\csromanfn@do{#1}{#2}{#3}{#4}}%
      \global\csromanfn@nexttoks=\expandafter{\the\toks@}%
      \global\advance\csromanfn@nextn\@ne
    \else
      \global\csromanfn@htsum=\csromanfn@ht
      \toks@=\expandafter{\the\csromanfn@pagetoks\csromanfn@do{#1}{#2}{#3}{#4}}%
      \global\csromanfn@pagetoks=\expandafter{\the\toks@}%
      \global\advance\csromanfn@pagen\@ne
    \fi
  \fi
}

\def\csromanfn@buildfootins{%
  \csromanfn@htheld=\dimexpr\ht\footins+\dp\footins\relax
  \global\csromanfn@pagetoks{}%
  \global\csromanfn@nexttoks{}%
  \global\csromanfn@htsum=\z@
  \global\csromanfn@pagen=\z@
  \global\csromanfn@nextn=\z@
  \csromanfn@holdingfalse
  \let\csromanfn@do\csromanfn@partition
  \the\csromanfn@toks
  \ifnum\csromanfn@pagen>\z@
    \global\setbox\footins=\vbox\bgroup
      \reset@font\footnotesize
      \frenchspacing
      \offinterlineskip
      \floatingpenalty\@MM
      \global\csromanfn@pending=\z@
      \global\csromanfn@mode=\z@
      \global\csromanfn@anyrowfalse
      % Numbered first, then * / + — keep pending short cells across the
      % pass boundary so e.g. short ¹ + short * share a 2-col row.
      \let\csromanfn@do\csromanfn@walknumbered
      \the\csromanfn@pagetoks
      \let\csromanfn@do\csromanfn@walksymbol
      \the\csromanfn@pagetoks
      \csromanfn@emitpending
    \egroup
  \else
    \global\setbox\footins=\box\voidb@x
  \fi
  \global\csromanfn@toks=\expandafter{\the\csromanfn@nexttoks}%
  \global\csromanfn@n=\csromanfn@nextn
}

% Same half/third cuts for numbered and apparatus marks.
% Sets \@tempswa (long) and \csromanfn@class; does not append.
\def\csromanfn@classify{%
  \ifdim\csromanfn@nat>\csromanfn@halfw
    \@tempswatrue
    \def\csromanfn@class{2}%
  \else
    \@tempswafalse
    \ifdim\csromanfn@nat>\csromanfn@thirdw
      \def\csromanfn@class{1}%
    \else
      \def\csromanfn@class{0}%
    \fi
  \fi
}

\long\def\csromanfn@footnotetext#1{%
  \global\advance\csromanfn@n\@ne
  \edef\csromanfn@mark{\@thefnmark}%
  \settowidth\csromanfn@nat{%
    \reset@font\footnotesize
    \frenchspacing
    \hbox{\normalfont\@thefnmark\hspace{\footnotemargin}#1}%
  }%
  \csromanfn@classify
  \setbox\csromanfn@box=\vbox{%
    \reset@font\footnotesize
    \frenchspacing
    \if@tempswa
      \hsize\columnwidth
      \columnwidth\columnwidth
      \linewidth\columnwidth
    \else
      \ifnum\csromanfn@class=\z@
        \hsize\csromanfn@thirdw
        \columnwidth\csromanfn@thirdw
        \linewidth\csromanfn@thirdw
      \else
        \hsize\csromanfn@halfw
        \columnwidth\csromanfn@halfw
        \linewidth\csromanfn@halfw
      \fi
    \fi
    \@parboxrestore
    \protected@edef\@currentlabel{%
      \csname p@footnote\endcsname\@thefnmark
    }%
    \@makefntext{%
      \rule\z@\footnotesep\ignorespaces#1\@finalstrut\strutbox}%
    \par
  }%
  \csromanfn@ht=\dimexpr\ht\csromanfn@box+\dp\csromanfn@box\relax
  \if@tempswa
    \csromanfn@reserved=\csromanfn@ht
  \else
    \ifnum\csromanfn@class=\z@
      \csromanfn@reserved=\dimexpr.34\csromanfn@ht\relax
    \else
      \csromanfn@reserved=\dimexpr.5\csromanfn@ht\relax
    \fi
  \fi
  % Freeze mark/class/reserved now — \csromanfn@class is reused for later notes.
  \begingroup
    \edef\x{\endgroup
      \noexpand\csromanfn@append
        {\csromanfn@mark}%
        {\unexpanded{#1}}%
        {\csromanfn@class}%
        {\the\csromanfn@reserved}%
    }%
  \x
  \insert\footins{%
    \floatingpenalty\@MM
    \vbox to\csromanfn@reserved{}%
  }%
}

\def\csromanfn@makecol{%
  \ifnum\csromanfn@n>\z@
    \csromanfn@buildfootins
  \fi
  \csromanfn@saved@makecol
  \csromanfn@clearpending
}

\AtBeginDocument{%
  \csromanfn@halfw=\dimexpr(\columnwidth-\csromanfn@colsep)/2\relax
  \csromanfn@thirdw=\dimexpr(\columnwidth-2\csromanfn@colsep)/3\relax
  \csromanfn@colw=\csromanfn@halfw
  \csromanfn@reset
  \let\csromanfn@saved@footnotetext\@footnotetext
  \let\csromanfn@saved@makecol\@makecol
  \let\@footnotetext\csromanfn@footnotetext
  \let\@makecol\csromanfn@makecol
}
