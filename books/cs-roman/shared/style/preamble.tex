\RequirePackage{iftex}
\RequireLuaTeX

% Reading / printing builds define a geometry flag before inputting this file.
% Printing scale s = 165mm/499bp ≈ 0.937308 (as 937308/1000000 for \dimexpr).
\ifdefined\csromanuseprintinggeometry
  \def\csromanprintingscalenum{937308}%
  \def\csromanprintingscaleden{1000000}%
  \def\csromanprintingscale{0.937308}%
  \input{shared/style/pagegeometry-printing}
\else
  \def\csromanprintingscalenum{1}%
  \def\csromanprintingscaleden{1}%
  \def\csromanprintingscale{1}%
  \ifdefined\csromanusereadinggeometry
    \input{shared/style/pagegeometry-reading}
  \else
    \input{shared/style/pagegeometry}
  \fi
\fi

\usepackage[
  main=thai,
  english,
  provide=*,
]{babel}

% CS Roman: คำศัพท์บาลีพิมพ์เว้นระหว่างคำ (ไม่ติดกันแบบไทยทั่วไป)
% จึงตัดบรรทัดได้เฉพาะที่ช่องว่าง — ไม่อนุญาตให้ตัดกลางคำ
\babelprovide[hyphenrules=nohyphenation]{thai}

% Body Scale: reading/sync 1.1; printing 1.1×s ≈ 1.031038.
\ifdefined\csromanuseprintinggeometry
  \def\csromanfontscale{1.031038}%
\else
  \def\csromanfontscale{1.1}%
\fi

\babelfont{rm}[
  Path           = shared/fonts/,
  Extension      = .ttf,
  UprightFont    = *-Regular,
  BoldFont       = *-Bold,
  ItalicFont     = *-Italic,
  BoldItalicFont = *-BoldItalic,
  Ligatures      = TeX,
  % Source body ≈12 pt VZTime; Sarabun 11pt × 1.1 ≈ 12.1 pt (printing × s).
  % Tuned so prose median line_ratio stays in ~0.95–1.05 vs CS Roman.
  Scale          = \csromanfontscale,
  % Font-load interword baseline only (fontspec cannot change WordSpace later).
  % Edition / volume absolute target: DEFAULT_LAYOUT / layout.json ``word_space``;
  % generator overwrites this via \csromanlayoutapply → \spaceskip scale.
  WordSpace      = 3.5,
]{Sarabun}

% Thai stack (ิ/ี + ํ, ฺ, ุ/ู) needs more leading than Latin; tight lead looks uneven.
% memoir 11pt \baselineskip ≈ 13.6 pt; × 1.5 ≈ 20.4 pt (edition default from 01Vin01).
% Defaults below match DEFAULT_LAYOUT in scripts/cs_roman_segments.py; body.tex
% may re-apply via \csromanlayoutapply from layout.json.
\linespread{1.5}\selectfont
\normalsize
\newlength{\tipitakahang}
\setlength{\tipitakahang}{\dimexpr21.6pt*\csromanprintingscalenum/\csromanprintingscaleden\relax}
\setlength{\parindent}{\tipitakahang}
% Body vertical rhythm:
%   \tipitakaparskip  — gap between prose paragraphs (edition default 5pt)
%   \gathastanzaskip  — target baseline gap between gāthā บท (default 30pt)
\newlength{\tipitakaparskip}
\newlength{\gathastanzaskip}
\setlength{\tipitakaparskip}{\dimexpr5pt*\csromanprintingscalenum/\csromanprintingscaleden\relax plus 1pt minus 0.5pt}
\setlength{\gathastanzaskip}{\dimexpr30pt*\csromanprintingscalenum/\csromanprintingscaleden\relax}
\setlength{\parskip}{\tipitakaparskip}
\setlength{\emergencystretch}{2.5em}
\frenchspacing
% Ragged bottom: page-sync leaves variable leftover space; flushbottom would
% stretch heading gaps on short pages (especially the opening title stack).
% Residual fill is limited to \vspace{0pt plus 2\baselineskip} in \csromanpage.
\raggedbottom
% Avoid a single line of a paragraph stranded across a page break
% (club = first line alone at page foot; widow = last line alone at page head).
% 10000 ≈ forbidden. In sync mode breaks are forced via \csromanpage anyway;
% in reading mode this still discourages club/widow lines. Oversized body→rule
% gaps on footnote pages are handled by packing credit in footnotes.tex (not by
% relaxing these penalties — see research/clubwidow_penalty_tuning.md).
\clubpenalty=10000
\widowpenalty=10000

\usepackage{microtype}
\usepackage{xcolor}
\usepackage{graphicx}
\usepackage[bottom,hang,stable,perpage]{footmisc}
\setlength{\footnotemargin}{0.85em}
% Extra headroom for Thai multi-level marks above the footnote baseline.
\setlength{\footnotesep}{\dimexpr8.5pt*\csromanprintingscalenum/\csromanprintingscaleden\relax}
% Gap from last body line to footnote rule: 50% above LaTeX default
% (\bigskipamount = 12pt plus 4pt minus 4pt → 18pt plus 6pt minus 6pt).
% \footnoterule still pulls the rule up by 3pt into this glue.
\skip\footins=1.5\bigskipamount
\usepackage{needspace}
% Reading/printing \csromanfolio (works inside gāthā \vtop; sync builds unused).
\usepackage{marginnote}

\usepackage[
  unicode=true,
  bookmarks=true,
  hidelinks,
  pdfencoding=auto,
  % No page.N anchors: roman front matter and arabic body both use ``1''.
  pageanchor=false,
  % Link TOC titles *and* page numbers to contentsline dest (#4 / mtk.*).
  % Without this, reading-mode \pageref page digits make their own (broken)
  % page-anchor links and jump to the start of the PDF.
  linktoc=all,
]{hyperref}
\usepackage{bookmark}

\usepackage{memhfixc}

\input{shared/style/book-macros}
\input{shared/style/footnotes}

\settocdepth{paragraph}
\maxtocdepth{paragraph}
\setsecnumdepth{none}
