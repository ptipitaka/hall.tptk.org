% Thai stack (ิ/ี + ํ, ฺ, ุ/ู) needs more leading than Latin; tight lead looks uneven.
% memoir 11pt \baselineskip ≈ 13.6 pt; × 1.25 ≈ 17.0 pt (was 1.152 / ~15.7 pt).

\normalsize

% Body vertical rhythm (measured para gap ≈23 pt ⇒ +≈6.3 pt over body lead):
%   \tipitakaparskip  — gap between prose paragraphs
%   \gathastanzaskip  — gap after each gāthā บท (defaults equal to parskip)
\newlength{\tipitakaparskip}
\newlength{\gathastanzaskip}
\setlength{\tipitakaparskip}{6.3pt plus 1pt minus 0.5pt}
\setlength{\gathastanzaskip}{\tipitakaparskip}
\setlength{\parskip}{\tipitakaparskip}
\setlength{\emergencystretch}{2.5em}
\frenchspacing
